###############################  Model ###############################



###############################  Sets  ###############################

set CLIENTS;
set MAGASINS;
set FLUIDE;

################### Variables ###################

var c{i in FLUIDE, j in CLIENTS, k in MAGASINS}, >= 0;
var choisie{j in CLIENTS, k in MAGASINS}, binary;

## To declare integer variables : 
# var Q{i in CIMENTS}, integer; 

## To declare binary variables : 
# var Q{i in CIMENTS}, binary; 


###################  Constants: Data to load   #########################

param commande{i in CLIENTS, j in FLUIDE};
param stocks{k in MAGASINS, j in FLUIDE};
param couts{k in MAGASINS, i in FLUIDE};
param cf{i in CLIENTS, k in MAGASINS};
param cv{i in CLIENTS, k in MAGASINS};
param BigM;

################### Constraints ###################

s.t. Respectcolonnes{i in FLUIDE, j in CLIENTS}:
  sum{k in MAGASINS} c[i,j, k] = commande[j,i];
s.t. RespectStocks{i in FLUIDE, k in MAGASINS} :
  sum{j in CLIENTS} c[i,j,k] <= stocks[k,i]; 
s.t. RespectSynchrosup{j in CLIENTS, k in MAGASINS}:
  choisie[j,k] * BigM >= sum{i in FLUIDE} c[i,j,k];
s.t. RespectSynchroinf{j in CLIENTS, k in MAGASINS}:
  choisie[j,k] <= sum{i in FLUIDE} c[i,j,k];

###### Objective ######

minimize cout: 
		sum{i in FLUIDE, j in CLIENTS, k in MAGASINS} (c[i,j,k] * cv[j,k]+ c[i,j,k]*couts[k,i]) + (sum{j in CLIENTS, k in MAGASINS} choisie[j,k]*cf[j,k]); 
    ##cout unitaire entre plus en jeu, ça marche 

end;
